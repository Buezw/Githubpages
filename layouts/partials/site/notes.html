{{- /* 收集 data/notebook.yaml 里列出的文件夹中的所有笔记，按日期从新到旧。 返回：slice of dict{ page, name, area, dir } */ -}} {{- $nb := site.Data.notebook -}} {{- $pages := slice -}} {{- with site.GetPage "/docs" -}} {{- range $p := .RegularPagesRecursive -}} {{- if and $p.File (not $p.Date.IsZero) -}} {{- $hit := false -}} {{- range $nb.sources -}} {{- if hasPrefix $p.File.Dir (printf "%s/" .dir) -}}{{- $hit = true -}}{{- end -}} {{- end -}} {{- if $hit -}}{{- $pages = $pages | append $p -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $items := slice -}} {{- range $p := sort $pages "Date" "desc" -}} {{- $src := dict -}} {{- range $nb.sources -}} {{- if hasPrefix $p.File.Dir (printf "%s/" .dir) -}}{{- $src = . -}}{{- end -}} {{- end -}} {{- $items = $items | append (dict "page" $p "name" $src.name "area" $src.area "dir" $src.dir) -}} {{- end -}} {{- return $items -}}
